function countUp(el){
  if(el.getAttribute('data-done')) return;
  el.setAttribute('data-done','1');
  var n = +el.getAttribute('data-n');
  var pre = el.getAttribute('data-pre') || '';
  var suf = el.getAttribute('data-suf') || '';
  var t0 = performance.now(), dur = 1300;
  function step(t){
    var p = Math.min(1, (t - t0) / dur);
    el.textContent = pre + Math.round(n * (1 - Math.pow(1 - p, 3))) + suf;
    if(p < 1) requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

      else if(ratio >= .5) add(on, 8, 'صور بوصف ALT (' + altOk + '/' + imgs.length + ')', 'part', 'نصف الصور بدون ALT. أكمل الباقي.');
      else add(on, 0, 'صور بوصف ALT (' + altOk + '/' + imgs.length + ')', 'fail', 'معظم الصور بدون ALT. فرص مهدرة في بحث الصور.');
    }
    if(words >= 300) add(on, 10, 'حجم المحتوى (' + words + ' كلمة)', 'pass', '');
    else if(words >= 120) add(on, 5, 'حجم المحتوى (' + words + ' كلمة)', 'part', 'المحتوى نحيف. الصفحات الفقيرة نصيًا يصعب تنافسها.');
    else add(on, 0, 'حجم المحتوى (' + words + ' كلمة)', 'fail', 'محتوى نحيف جدًا. يحتاج Google إلى مادة يفهم منها الصفحة.');
    if(internal >= 3) add(on, 10, 'روابط داخلية (' + internal + ')', 'pass', '');
    else if(internal >= 1) add(on, 5, 'روابط داخلية (' + internal + ')', 'part', 'قليلة. اربط الصفحات ببعضها لتوزيع القوة.');
    else add(on, 0, 'روابط داخلية', 'fail', 'لا توجد. الصفحة معزولة عن بقية الموقع.');

    /* Technical */
    if(urlObj.protocol === 'https:') add(tech, 20, 'اتصال آمن HTTPS', 'pass', '');
    else add(tech, 0, 'اتصال آمن HTTPS', 'fail', 'لا يزال HTTP. المتصفحات تحذر الزوار.');
    if(viewport) add(tech, 15, 'وسم Viewport', 'pass', '');
    else add(tech, 0, 'وسم Viewport', 'fail', 'الموقع غير متوافق مع الجوال، والفهرسة تتبع الجوال أولًا.');
    if(charset) add(tech, 5, 'ترميز الأحرف (Charset)', 'pass', '');
    else add(tech, 0, 'ترميز الأحرف', 'fail', 'بدون Charset قد تظهر الحروف كرموز.');
    if(doc.doctype) add(tech, 5, 'إعلان Doctype', 'pass', '');
    else add(tech, 0, 'إعلان Doctype', 'fail', 'يدخل المتصفح وضع التوافق القديم.');
    if(lang) add(tech, 5, 'لغة الصفحة (lang)', 'pass', '');
    else add(tech, 0, 'لغة الصفحة (lang)', 'fail', 'حدد اللغة ليصل Google إلى الجمهور الصحيح.');
    if(canonical) add(tech, 10, 'وسم Canonical', 'pass', '');
    else add(tech, 0, 'وسم Canonical', 'fail', 'بدونه تتنافس نسخ الصفحة على الترتيب.');
    if(robots.indexOf('noindex') === -1) add(tech, 10, 'لا يوجد منع فهرسة', 'pass', '');
    else add(tech, 0, 'منع الفهرسة (noindex)', 'fail', 'لن تظهر صفحتك في Google أصلًا. هذا أول ما يجب إصلاحه.');
    if(jsonld > 0) add(tech, 15, 'بيانات منظمة Schema (' + jsonld + ')', 'pass', '');
    else add(tech, 0, 'بيانات منظمة Schema', 'fail', 'ستخسر النجوم والأسعار في النتائج.');
    if(sizeKB < 300) add(tech, 10, 'حجم الصفحة (' + sizeKB + ' KB)', 'pass', '');
    else if(sizeKB < 600) add(tech, 5, 'حجم الصفحة (' + sizeKB + ' KB)', 'part', 'ثقيلة وتؤخر أول ظهور.');
    else add(tech, 0, 'حجم الصفحة (' + sizeKB + ' KB)', 'fail', 'ضخم. يقتل السرعة، والسرعة عامل ترتيب.');
    if(favicon) add(tech, 5, 'أيقونة الموقع (Favicon)', 'pass', '');
    else add(tech, 0, 'أيقونة الموقع (Favicon)', 'fail', 'تظهر في نتائج الجوال وتبويبات المتصفح.');

    /* Off-Page */
    if(social > 0) add(off, 35, 'حسابات تواصل اجتماعي مرتبطة (' + social + ')', 'pass', '');
    else add(off, 0, 'حسابات تواصل اجتماعي مرتبطة', 'fail', 'لا توجد روابط لحساباتك. ربطها يبني هوية الموقع.');
    if(ogT && ogI) add(off, 25, 'جاهزية المشاركة (Open Graph)', 'pass', '');
    else if(ogT || ogI) add(off, 12, 'جاهزية المشاركة (OG ناقص)', 'part', 'ناقصة. أكملها لشكل مشاركات احترافي.');
    else add(off, 0, 'جاهزية المشاركة (OG)', 'fail', 'أي مشاركة ستظهر مكسورة بدون وسوم OG.');
    if(external > 0) add(off, 15, 'إشارة إلى مصادر خارجية (' + external + ')', 'pass', '');
    else add(off, 0, 'إشارة إلى مصادر خارجية', 'fail', 'لا توجد إحالات، وهي جزء من بناء الثقة.');
    if(contact > 0) add(off, 15, 'قنوات تواصل واضحة', 'pass', '');
    else add(off, 0, 'قنوات تواصل واضحة', 'fail', 'لا توجد وسيلة تواصل في الصفحة.');
    if(twCard) add(off, 10, 'بطاقة Twitter Card', 'pass', '');
    else add(off, 0, 'بطاقة Twitter Card', 'fail', 'سطر واحد يضبط شكل المشاركات على X.');

    var overall = Math.round(on.score * .4 + tech.score * .4 + off.score * .2);
    return {on:on, tech:tech, off:off, overall:overall};
  }

  function ringHTML(id, label, sub, big){
    var sz = big ? 140 : 110, c = big ? 70 : 55, r = big ? 59 : 46;
    return '<div class="sr-item"><div class="sr-ring' + (big ? ' big' : '') + '" id="' + id + '">' +
      '<svg viewBox="0 0 ' + sz + ' ' + sz + '">' +
      '<circle class="sr-track" cx="' + c + '" cy="' + c + '" r="' + r + '"/>' +
      '<circle class="sr-prog" cx="' + c + '" cy="' + c + '" r="' + r + '"/></svg>' +
      '<span class="sr-val">0</span></div>' +
      '<label>' + label + '</label><span class="sr-sub">' + sub + '</span></div>';
  }
  function fillRing(id, pct){
    var el = document.getElementById(id);
    if(!el) return;
    var circ = el.querySelector('.sr-prog');
    var R = el.classList.contains('big') ? 59 : 46;
    var C = 2 * Math.PI * R;
    circ.setAttribute('stroke', pct >= 80 ? '#5DCA7C' : (pct >= 50 ? '#F59E0B' : '#EF4444'));
    circ.setAttribute('stroke-dasharray', C);
    circ.style.strokeDashoffset = C;
    requestAnimationFrame(function(){
      requestAnimationFrame(function(){
        circ.style.strokeDashoffset = C * (1 - pct / 100);
      });
    });
    el.querySelector('.sr-val').textContent = Math.round(pct);
  }
  function colHTML(titleTxt, cat){
    var h = '<div class="check-col"><h4>' + titleTxt + '<span class="cc-score">' + Math.round(cat.score) + '/100</span></h4>';
    for(var i = 0; i < cat.items.length; i++){
      var it = cat.items[i];
      var ic = it.s === 'pass' ? '✓' : (it.s === 'part' ? '~' : '✗');
      h += '<div class="chk ' + it.s + '"><span class="ic">' + ic + '</span><span class="t">' + it.t +
           (it.hint ? '<span class="hint">' + it.hint + '</span>' : '') + '</span></div>';
    }
    return h + '</div>';
  }
  function verdictText(v){
    if(v >= 85) return 'الأساس سليم. إذا لم تتحرك الأرقام بالقوة نفسها فالفجوة في الاستراتيجية لا التقنية.';
    if(v >= 65) return 'أساس جيد، لكن الفجوات أعلاه تأكل من ترتيبك. ومعظمها قابل للإصلاح السريع.';
    if(v >= 40) return 'يرى Google الموقع نصف مكتمل. كل ✗ هنا باب كان يمكن أن يجلب زوارًا.';
    return 'Google بالكاد يرى الموقع. الخبر الجيد أن البدء من هنا أسهل، وكل تحسين سيظهر بسرعة.';
  }

  function show(el){ el.hidden = false; }
  function hide(el){ el.hidden = true; }

  form.addEventListener('submit', function(ev){
    ev.preventDefault();
    var raw = input.value.trim();
    var urlObj = null;
    if(raw){
      if(!/^https?:\/\//i.test(raw)) raw = 'https://' + raw;
      try{
        var u = new URL(raw);
        if(u.hostname.indexOf('.') !== -1) urlObj = u;
      }catch(e){}
    }
    if(!urlObj){
      input.focus();
      input.placeholder = 'اكتب رابطًا صحيحًا مثل example.com';
      return;
    }

    btn.disabled = true;
    hide(errBox); hide(resBox); show(statusBox);
    var stages = ['جارٍ الاتصال بالموقع…', 'جارٍ قراءة الصفحة…', 'جارٍ فحص العناصر…', 'جارٍ حساب النسب…'];
    var st = 0;
    statusText.textContent = stages[0];
    var stTimer = setInterval(function(){
      st = Math.min(st + 1, stages.length - 1);
      statusText.textContent = stages[st];
    }, 1600);

    fetchHTML(urlObj.href).then(function(html){
      clearInterval(stTimer);
      var doc = new DOMParser().parseFromString(html, 'text/html');
      var result = analyze(doc, html.length, urlObj);

      urlChip.textContent = urlObj.hostname.replace(/^www\./, '');
      ringsRow.innerHTML =
        ringHTML('rgOverall', 'النتيجة الإجمالية', 'On-Page 40% + Technical 40% + Off-Page 20%', true) +
        ringHTML('rgOn', 'On-Page SEO', 'المحتوى والعناصر الظاهرة', false) +
        ringHTML('rgTech', 'Technical SEO', 'الأساس التقني للفهرسة', false) +
        ringHTML('rgOff', 'Off-Page', 'مؤشرات أولية*', false);
      verdictEl.textContent = verdictText(result.overall);
      checksGrid.innerHTML =
        colHTML('On-Page SEO', result.on) +
        colHTML('Technical SEO', result.tech) +
        colHTML('Off-Page', result.off);

      hide(statusBox);
      show(resBox);
      fillRing('rgOverall', result.overall);
      fillRing('rgOn', result.on.score);
      fillRing('rgTech', result.tech.score);
      fillRing('rgOff', result.off.score);
      resBox.scrollIntoView({behavior:'smooth', block:'start'});
      btn.disabled = false;
    }).catch(function(){
      clearInterval(stTimer);
      hide(statusBox);
      show(errBox);
      errBox.scrollIntoView({behavior:'smooth', block:'center'});
      btn.disabled = false;
    });
  });
})();
